src/components/ShoppingCart.jsx Composite Parent Component (Tasks 2 & 5) 
import { useState } from 'react';
import { products as initialProducts } from '../data/products';
import ProductList from './ProductList';
import CartSidebar from './CartSidebar';

function ShoppingCart() {
  // Task 2: State manages an array of cart items
  const [cartItems, setCartItems] = useState([]);

  // Task 2: Add to cart with quantity increment
  const handleAddToCart = (product) => {
    setCartItems((prevItems) => {
      const existing = prevItems.find((item) => item.id === product.id);
      if (existing) {
        return prevItems.map((item) =>
          item.id === product.id ? { ...item, quantity: item.quantity + 1 } : item
        );
      }
      return [...prevItems, { ...product, quantity: 1 }];
    });
  };

  // Task 2: Remove item from cart state
  const handleRemoveFromCart = (productId) => {
    setCartItems((prevItems) => prevItems.filter((item) => item.id !== productId));
  };

  // Task 4: Quantity controls (+ and - per item)
  const handleUpdateQuantity = (productId, delta) => {
    setCartItems((prevItems) =>
      prevItems
        .map((item) => {
          if (item.id === productId) {
            const nextQty = item.quantity + delta;
            return nextQty > 0 ? { ...item, quantity: nextQty } : null;
          }
          return item;
        })
        .filter(Boolean)
    );
  };

  const handleClearCart = () => setCartItems([]);

  return (
    <div className="shopping-cart-container">
      {/* Task 1: Product list component */}
      <ProductList products={initialProducts} onAddToCart={handleAddToCart} />

      {/* Task 3: Cart sidebar component */}
      <CartSidebar
        cartItems={cartItems}
        onUpdateQuantity={handleUpdateQuantity}
        onRemoveFromCart={handleRemoveFromCart}
        onClearCart={handleClearCart}
      />
    </div>
  );
}

export default ShoppingCart;
src/components/CartSidebar.jsx Sidebar with Total Price Computed from State (Task 3) 
import CartItem from './CartItem';

function CartSidebar({ cartItems, onUpdateQuantity, onRemoveFromCart, onClearCart }) {
  // Task 3: Dynamically computed values derived directly from state
  const totalItems = cartItems.reduce((sum, item) => sum + item.quantity, 0);
  const totalPrice = cartItems.reduce((sum, item) => sum + item.price * item.quantity, 0);

  return (
    <aside className="cart-sidebar">
      <div className="cart-header">
        <h2>Your Cart</h2>
        <span className="cart-badge">{totalItems} items</span>
      </div>

      {cartItems.length === 0 ? (
        <p className="empty-text">Your cart is empty.</p>
      ) : (
        <>
          <ul className="cart-items-list">
            {cartItems.map((item) => (
              <CartItem
                key={item.id}
                item={item}
                onUpdateQuantity={onUpdateQuantity}
                onRemoveFromCart={onRemoveFromCart}
              />
            ))}
          </ul>

          <div className="cart-total-box">
            <span>Total Price:</span>
            <span className="total-amount">${totalPrice.toFixed(2)}</span>
          </div>
        </>
      )}
    </aside>
  );
}

export default CartSidebar;
src/components/CartItem.jsx Quantity Controls + Remove Button (Tasks 2 & 4) 
function CartItem({ item, onUpdateQuantity, onRemoveFromCart }) {
  const { id, name, price, image, quantity } = item;

  return (
    <li className="cart-item">
      <img src={image} alt={name} className="cart-item-img" />
      <div className="cart-item-details">
        <h4>{name}</h4>
        <span>${price.toFixed(2)} each</span>

        {/* Task 4: Quantity controls */}
        <div className="qty-group">
          <button onClick={() => onUpdateQuantity(id, -1)}>−</button>
          <span>{quantity}</span>
          <button onClick={() => onUpdateQuantity(id, 1)}>+</button>
        </div>
      </div>

      {/* Task 2: Remove button */}
      <button className="btn-remove" onClick={() => onRemoveFromCart(id)}>✕</button>
    </li>
  );
}

export default CartItem;

function ProductList({ products, onAddToCart }) {
  return (
    <section className="product-list">
      <h2>Available Products ({products.length})</h2>
      <div className="products-grid">
        {products.map((product) => (
          <ProductCard
            key={product.id}
            product={product}
            onAddToCart={onAddToCart}
          />
        ))}
      </div>
    </section>
  );
}

export default ProductList;
src/components/ProductCard.jsx Name, Price, Image URL + Add to Cart Button (Tasks 1 & 2) 
function ProductCard({ product, onAddToCart }) {
  const { id, name, price, image, category, description } = product;

  return (
    <div className="product-card">
      <img src={image} alt={name} className="product-img" />
      <div className="product-content">
        <h3>{name}</h3>
        <p>{description}</p>
        <div className="product-footer">
          <span className="price">${price.toFixed(2)}</span>
          {/* Task 2: Add to Cart button */}
          <button onClick={() => onAddToCart(product)}>Add to Cart</button>
        </div>
      </div>
    </div>
  );
}

export default ProductCard;
