Conditional Rendering in Notification Lists

Task Requirement: Given an array of notification objects [{ id, type, message }], render each as a styled item. Show a checkmark icon (✅) for type === 'success', a warning icon (⚠️) for type === 'warning', and a default info icon (ℹ️) otherwise. Use conditional rendering.

// Notification Data Array
const notifications = [
  { id: 1, type: 'success', message: 'Profile updated successfully' },
  { id: 2, type: 'warning', message: 'Password expires in 3 days' },
  { id: 3, type: 'info', message: 'New lecture materials available' },
];

// Presentational Item Component with Nested Ternary Conditional Icon
function NotificationItem({ type, message }) {
  const icon = type === 'success' 
    ? '✅' 
    : type === 'warning' 
    ? '⚠️' 
    : 'ℹ️';

  return (
    <li className={`notification-item item-${type}`}>
      <span className="notif-icon">{icon}</span>
      <span className="notif-message">{message}</span>
    </li>
  );
}

// Container Component Rendering Dynamic List
function NotificationList() {
  return (
    <ul className="notification-ul">
      {notifications.map(n => (
        <NotificationItem 
          key={n.id} 
          type={n.type} 
          message={n.message} 
        />
      ))}
    </ul>
  );
}
