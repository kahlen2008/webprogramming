// src/components/ProjectCard.jsx
function ProjectCard({ title, description, techStack, githubLink, demoLink, caseStudyLink, domain }) {
  return (
    <div className="project-card">
      <div className="project-card-top">
        <div className="project-header-row">
          <h3 className="project-title">{title}</h3>
          {domain && <span className="project-domain-tag">{domain}</span>}
        </div>
        <p className="project-desc">{description}</p>
        <div className="tech-stack-row">
          {techStack.map((tech) => (
            <span key={tech} className="tech-badge">
              {tech}
            </span>
          ))}
        </div>
      </div>

      <div className="project-card-footer">
        {caseStudyLink ? (
          <a
            href={caseStudyLink}
            target="_blank"
            rel="noopener noreferrer"
            className="btn-card-link btn-study"
          >
            Explore Case Study ↗
          </a>
        ) : (
          <>
            {githubLink && (
              <a
                href={githubLink}
                target="_blank"
                rel="noopener noreferrer"
                className="btn-card-link btn-repo"
              >
                GitHub Repo
              </a>
            )}
            {demoLink && (
              <a
                href={demoLink}
                target="_blank"
                rel="noopener noreferrer"
                className="btn-card-link btn-demo"
              >
                Live Demo ↗
              </a>
            )}
          </>
        )}
      </div>
    </div>
  );
}

export default ProjectCard;

// src/components/ProjectList.jsx
import ProjectCard from './ProjectCard';

function ProjectList({ projects }) {
  if (projects.length === 0) {
    return (
      <div className="empty-state">
        <p>No projects match the selected technology filter.</p>
      </div>
    );
  }

  return (
    <div className="project-cards-grid">
      {projects.map((project) => (
        <ProjectCard
          key={project.id}
          title={project.title}
          description={project.description}
          techStack={project.techStack}
          githubLink={project.githubLink}
          demoLink={project.demoLink}
          caseStudyLink={project.caseStudyLink}
          domain={project.domain}
        />
      ))}
    </div>
  );
}

export default ProjectList;
src/components/FilterBar.jsx 
// src/components/FilterBar.jsx
function FilterBar({ categories, activeFilter, onSelectFilter }) {
  return (
    <div className="filter-buttons-wrapper">
      {categories.map((category) => (
        <button
          key={category}
          type="button"
          className={`filter-pill ${activeFilter === category ? 'active' : ''}`}
          onClick={() => onSelectFilter(category)}
        >
          {category}
        </button>
      ))}
    </div>
  );
}

export default FilterBar;

src/App.jsx 
// src/App.jsx
import { useState } from 'react';
import { projectsData } from './data/projectsData';
import ProjectList from './components/ProjectList';
import FilterBar from './components/FilterBar';

// Extract all unique technologies dynamically across all projects
const CATEGORIES = [
  'All',
  ...Array.from(new Set(projectsData.flatMap((p) => p.techStack)))
];

function App() {
  const [activeFilter, setActiveFilter] = useState('All');

  // Filter logic matching project techStack array
  const filteredProjects = activeFilter === 'All'
    ? projectsData
    : projectsData.filter((p) => p.techStack.includes(activeFilter));

  return (
    <div className="portfolio-gallery-app">
      <header className="gallery-header">
        <h1>Kahlen Ong · Project Portfolio</h1>
        <p>Filtered by technology using modular React components.</p>
      </header>

      <FilterBar
        categories={CATEGORIES}
        activeFilter={activeFilter}
        onSelectFilter={setActiveFilter}
      />

      <ProjectList projects={filteredProjects} />
    </div>
  );
}

export default App;

Specification verification against the Week 11_1 Homework 2 requirements.

Component Prop Contract Type Role & Responsibilities
ProjectCard { title, description, techStack, githubLink, demoLink, caseStudyLink } Object props Pure presentational leaf component. Renders project headers, text summary, mapped technology badges, and action buttons.
ProjectList { projects } Array of objects Maps over the filtered dataset using unique key={project.id}; presents an empty state fallback if the count is zero.
FilterBar { categories, activeFilter, onSelectFilter } Array, String, Fn Renders technology pill buttons, calculates matching items, and invokes parent state updates via callback parameter.
App (Container) None (Root) Root Component Manages activeFilter state, computes derived filtered arrays, and composes children.
