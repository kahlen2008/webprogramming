Component Composition

Assemble scalable UIs by composing focused, atomic components (e.g. Avatar + Badge into StudentCard) instead of maintaining bloated monolithic components.

StudentCard.jsx (Composed Architecture) Composition over Monoliths 
// 1. Atomic Avatar Component
function Avatar({ src, alt, size = 52 }) {
  return (
    <img 
      src={src} 
      alt={alt} 
      className="composed-avatar"
      style={{ width: size, height: size }} 
    />
  );
}

// 2. Atomic Badge Component
function StatusBadge({ text, color = '#3b82f6' }) {
  return (
    <span className="composed-badge" style={{ backgroundColor: color }}>
      {text}
    </span>
  );
}

// 3. Composed Container: StudentCard
function StudentCard({ name, studentId, major, year, avatarUrl, badgeColor }) {
  return (
    <div className="composed-student-card">
      <Avatar src={avatarUrl} alt={name} />
      <div className="composed-info">
        <h3>{name}</h3>
        <p className="meta-row">
          ID: <code>{studentId}</code> · {major}
        </p>
      </div>
      <StatusBadge text={`Year ${year}`} color={badgeColor} />
    </div>
  );
}

// 4. Composed in App
function App() {
  return (
    <div className="roster-list">
      <StudentCard 
        name="Kahlen Ong"
