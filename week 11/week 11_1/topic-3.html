Week 11_1 Hub Topic 03 · Props & Children

Props — Passing Data

Props flow unidirectionally from parent to child as read-only arguments. Destructuring allows default fallbacks, and props.children unlocks flexible composition slots.

Panel.jsx & ActionButton.jsx Destructuring & Default Props 
// 1. Button with destructuring and default prop values
function ActionButton({ 
  label = 'View Profile', 
  color = '#3b82f6', 
  disabled = false,
  onClick 
}) {
  return (
    <button 
      className="custom-btn" 
      style={{ backgroundColor: color }} 
      disabled={disabled}
      onClick={onClick}
    >
      {label}
    </button>
  );
}

// 2. Reusable Panel wrapper using props.children
function Panel({ title, children }) {
  return (
    <div className="panel-container">
      <h2 className="panel-header">{title}</h2>
      <div className="panel-body">{children}</div>
    </div>
  );
}

// 3. App passes custom & default props into children slot
function App() {
  return (
    <Panel title="Week 11_1 · Student Console">
      <p>Student: <strong>Kahlen Ong</strong> (ID: 202699471)</p>
      <div className="buttons-row">
        {/* 1: uses default label & color */}
        <ActionButton />
        {/* 2: custom label & green color */}
        <ActionButton label="Open Live App" color="#10b981" />
        {/* 3: custom red color & disabled */}
        <ActionButton label="Restricted Admin" color="#ef4444" disabled />
      </div>
    </Panel>
  );
}
