Lists & Conditional Rendering

Render dynamic collections with .map() bound to unique keys, and control JSX output via ternaries (? :), logical operators (&&), and early returns.

CoursePortal.jsx .map() + Unique Keys 
// Stable dataset with unique IDs
const courses = [
  { id: 'cs110', name: 'Web Programming (React)', credits: 3, completed: true },
  { id: 'cs120', name: 'Digital Logic Design', credits: 4, completed: true },
  { id: 'cs210', name: 'Data Structures', credits: 4, completed: false },
];

function CoursePortal({ isLoggedIn, username = 'Kahlen Ong' }) {
  // 1. Early Return Pattern
  if (!isLoggedIn) {
    return <p className="guest-alert">Please log in to inspect your curriculum.</p>;
  }

  return (
    <div className="course-portal">
      {/* 2. Ternary Operator */}
      <h2>
        {isLoggedIn ? `Enrolled: ${username}` : 'Guest Portal'}
      </h2>

      {/* 3. Logical && Operator */}
      {isLoggedIn && <span className="badge-status">Active Session</span>}

      {/* 4. List Rendering with Unique Key */}
      <ul className="courses-list">
        {courses.map(course => (
          <li key={course.id} className="course-item">
            <span>{course.name} ({course.credits} cr)</span>
            {course.completed ? '✅ Done' : '⏳ In Progress'}
          </li>
        ))}
      </ul>
    </div>
  );
}
