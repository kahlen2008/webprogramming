ProfileCard.jsx Component Definition & Usage 
// Definition: Destructuring props with default fallback for avatarUrl
function ProfileCard({ 
  name, 
  role, 
  email, 
  avatarUrl = 'https://via.placeholder.com/80' 
}) {
  return (
    <div style={{ border: '1px solid #ccc', borderRadius: 8, padding: 16, maxWidth: 300 }}>
      <img 
        src={avatarUrl} 
        alt={name} 
        style={{ borderRadius: '50%', width: 80, height: 80 }} 
      />
      <h2>{name}</h2>
      <p>{role}</p>
      <a href={`mailto:${email}`}>{email}</a>
    </div>
  );
}

// Usage in App:
<ProfileCard 
  name="Kahlen Ong" 
  role="Student" 
  email="kahlenongzhiqi@gmail.com" 
/>
