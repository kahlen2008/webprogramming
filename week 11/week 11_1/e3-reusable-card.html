// Reusable Container Component
function Card({ title, children }) {
  return (
    <div style={{ 
      border: '1px solid #bae6fd', 
      borderRadius: 10, 
      padding: '16px 20px', 
      marginBottom: 16 
    }}>
      <h3 style={{ marginBottom: 10 }}>{title}</h3>
      {children}
    </div>
  );
}

// Component Composition Usage
function App() {
  return (
    <>
      {/* Instance 1: Wrapping an Unordered List */}
      <Card title="Lecture Topics">
        <ul>
          <li>Components</li>
          <li>Props</li>
        </ul>
      </Card>

      {/* Instance 2: Wrapping a Paragraph */}
      <Card title="Homework">
        <p>Build a student portfolio component</p>
      </Card>
    </>
  );
}
