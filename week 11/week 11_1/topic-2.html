Functional Components

Components are plain JavaScript functions that return JSX, named with an uppercase first letter and utilizing Fragments <>...</> to return multiple elements cleanly.

StudentBadge.jsx & App.jsx Capitalized Name & Fragments 
// 1. Functional component returning multiple elements using <> Fragment
function StudentBadge({ name, role }) {
  return (
    <>
      <h3 className="badge-name">{name}</h3>
      <p className="badge-desc">{role}</p>
    </>
  );
}

// 2. Arrow function syntax (also valid)
const Header = () => <h2>Department Directory</h2>;

// 3. App orchestrates multiple instances (<StudentBadge />)
function App() {
  return (
    <div className="directory-board">
      <Header />
      <StudentBadge 
        name="Kahlen Ong" 
        role="Computer Engineering · Year 1" 
      />
      <StudentBadge 
        name="Aale Aaditi" 
        role="Computer Engineering · Year 2" 
      />
    </div>
  );
}
