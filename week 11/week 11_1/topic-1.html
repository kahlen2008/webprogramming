What is React & JSX?

React describes declarative UI states; JSX compiles HTML-like structures into native React.createElement() function calls.

App.jsx (JSX Syntax) Babel Compiles to React.createElement 
// Functional component returning a single root element
function App() {
  const student = 'Kahlen Ong';
  const major = 'Computer Engineering';
  const today = new Date().toLocaleDateString();

  return (
    <div className="portfolio-preview-card">
      <h1 className="welcome-heading">
        Welcome, {student}!
      </h1>
      <p className="student-meta">
        Major: <strong>{major}</strong> · Gachon University
      </p>
      <p className="session-date">
        Session Date: {today}
      </p>
      {/* Self-closing void element with alt attribute */}
      <img 
        src="https://api.dicebear.com/7.x/bottts/svg?seed=Kahlen" 
        alt="Kahlen Ong Student Avatar" 
        className="avatar-img" 
      />
    </div>
  );
}
