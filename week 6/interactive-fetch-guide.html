2. Handling a Promise (.then & .catch)
Promise State Stream
JavaScript (promise-chain.js) 
fetch(url)
  .then(response => {
    // 1. runs when fulfilled (network response arrived)
    return response.json();
  })
  .then(data => {
    // 2. runs with the parsed JSON data
    console.log(data);
  })
  .catch(error => {
    // 3. runs if rejected (network drop, DNS error)
    console.error('Failed:', error);
  });

4. Error Handling (The 404 Gotcha)
try / catch / finally
JavaScript (loadData.js) 
async function loadData(url) {
  try {
    const res = await fetch(url);

    // ⚠️ 404 does NOT throw! You must check this:
    if (!res.ok) throw new Error(`HTTP error: ${res.status}`);

    const data = await res.json();
    console.log('Success:', data);

  } catch (err) {
    console.error('Caught error:', err.message);

  } finally {
    console.log('finally: always runs!');
  }
}
